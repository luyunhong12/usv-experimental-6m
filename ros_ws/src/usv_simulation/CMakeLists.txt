cmake_minimum_required(VERSION 3.16)
project(usv_simulation)
find_package(ament_cmake REQUIRED)
find_package(usv_energy REQUIRED)
find_package(vrx_gz REQUIRED)
find_library(VRX_WAVES_LIBRARY Waves REQUIRED PATHS "${vrx_gz_DIR}/../../../lib")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/../../../dependencies/src/vrx/vrx_gz/src")
find_package(usv_physics REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(UsvHullHydrostatics SHARED src/HullHydrostatics.cc)
target_compile_features(UsvHullHydrostatics PUBLIC cxx_std_17)
target_compile_options(UsvHullHydrostatics PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O2>)
target_include_directories(UsvHullHydrostatics PRIVATE include)
target_link_libraries(UsvHullHydrostatics ${VRX_WAVES_LIBRARY} gz-sim8::core gz-plugin2::register gz-transport13::core)

target_link_libraries(UsvSteeringCommandFilter gz-sim8::core gz-plugin2::register gz-transport13::core)
add_library(UsvAzimuthPropulsion SHARED src/AzimuthPropulsion.cc)
target_compile_features(UsvAzimuthPropulsion PUBLIC cxx_std_17)
# Repeated geometric clipping is a real-time physics hot path.
target_compile_options(UsvAzimuthPropulsion PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O2>)
target_include_directories(UsvAzimuthPropulsion PRIVATE include ${usv_energy_INCLUDE_DIRS} ${usv_physics_INCLUDE_DIRS})
target_link_libraries(UsvAzimuthPropulsion ${VRX_WAVES_LIBRARY} gz-sim8::core gz-plugin2::register gz-transport13::core)
add_library(UsvAppendageHydrodynamics SHARED src/AppendageHydrodynamics.cc)
target_compile_features(UsvAppendageHydrodynamics PUBLIC cxx_std_17)
# 附体逐面水动力每个物理步执行；标准优化不改变步长、浮点规则或断言。
target_compile_options(UsvAppendageHydrodynamics PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O2>)
target_include_directories(UsvAppendageHydrodynamics PRIVATE include)
target_link_libraries(UsvAppendageHydrodynamics ${VRX_WAVES_LIBRARY} gz-sim8::core gz-plugin2::register gz-transport13::core)
install(TARGETS UsvAppendageHydrodynamics DESTINATION lib)
install(TARGETS UsvHullHydrostatics UsvSteeringCommandFilter UsvAzimuthPropulsion DESTINATION lib)
install(DIRECTORY worlds DESTINATION share/${PROJECT_NAME})
