cmake_minimum_required(VERSION 3.16)
project(usv_energy)
find_package(ament_cmake REQUIRED)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_export_include_directories(include)
if(BUILD_TESTING)
 enable_testing()
 add_executable(test_energy tests/test_energy.cc)
 target_compile_features(test_energy PUBLIC cxx_std_17)
 target_include_directories(test_energy PRIVATE include)
 add_test(NAME energy_conservation_and_faults COMMAND test_energy)
 add_executable(energy_scenario_report tests/energy_scenario_report.cc)
 target_compile_features(energy_scenario_report PUBLIC cxx_std_17)
 target_include_directories(energy_scenario_report PRIVATE include)
endif()
ament_package()
