cmake_minimum_required(VERSION 3.16)
project(usv_native_validation LANGUAGES CXX)
# Reuse production headers and existing assertions in an isolated instrumented build.
# This suite does not launch Gazebo or establish physical calibration.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(USV_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(gz-sim8 REQUIRED)
enable_testing()
set(USV_NATIVE_TESTS "energy;wrench_audit;appendages;steering_integral;resistance;added_mass;fuel_mass;propulsion;steering_reference;wave_kinematics;wave_pressure;hydrostatics" CACHE STRING "Native test targets to build and execute")
foreach(name IN LISTS USV_NATIVE_TESTS)
  if(name STREQUAL "energy")
    add_executable(test_energy "${USV_ROOT}/ros_ws/src/usv_energy/tests/test_energy.cc")
    target_include_directories(test_energy PRIVATE "${USV_ROOT}/ros_ws/src/usv_energy/include")
    add_test(NAME energy_conservation_and_faults COMMAND test_energy)
    continue()
  endif()
  add_executable(test_${name} "${USV_ROOT}/ros_ws/src/usv_simulation/tests/test_${name}.cc")
  target_include_directories(test_${name} PRIVATE "${USV_ROOT}/ros_ws/src/usv_simulation/include")
  target_link_libraries(test_${name} PRIVATE gz-sim8::core)
  add_test(NAME ${name} COMMAND test_${name})
endforeach()
